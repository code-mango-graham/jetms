-- =====================================================================
-- SCHOOL CALENDAR (admin-authored events, visible read-only to every role
-- via the shared topbar "School Calendar" link — see config/event.php)
-- =====================================================================

CREATE TABLE `tbl_event` (
  `event_id` int(11) NOT NULL AUTO_INCREMENT,
  `title` varchar(150) NOT NULL,
  `description` text DEFAULT NULL,
  `event_type` enum('Holiday','Exam','Meeting','Deadline','Other') NOT NULL DEFAULT 'Other',
  `start_date` date NOT NULL,
  `end_date` date DEFAULT NULL,
  `admin_id` int(11) NOT NULL,
  `admin_name` varchar(150) NOT NULL,
  `event_remarks` tinyint(1) NOT NULL DEFAULT 1,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`event_id`),
  KEY `fk_event_admin` (`admin_id`),
  KEY `idx_event_dates` (`start_date`,`end_date`),
  CONSTRAINT `fk_event_admin` FOREIGN KEY (`admin_id`) REFERENCES `tbl_admin` (`admin_id`)
  -- end_date is NULL for a single-day event; `idx_event_dates` supports the
  -- range query the calendar month-view uses to fetch events overlapping
  -- whatever date range is currently on screen
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =====================================================================
-- ATTENDANCE (gate scan log — see attendance_kiosk.html, no login required)
-- =====================================================================

CREATE TABLE `tbl_attendance_log` (
  `log_id` int(11) NOT NULL AUTO_INCREMENT,
  `student_id` int(11) NOT NULL,
  `lrn` varchar(20) NOT NULL,
  `student_name` varchar(200) NOT NULL,
  `log_type` enum('in','out') NOT NULL,
  `log_time` datetime NOT NULL DEFAULT current_timestamp(),
  `source` varchar(20) NOT NULL DEFAULT 'manual',
  PRIMARY KEY (`log_id`),
  KEY `fk_attendance_student` (`student_id`),
  CONSTRAINT `fk_attendance_student` FOREIGN KEY (`student_id`) REFERENCES `tbl_student` (`student_id`)
  -- in/out is toggled automatically per student per calendar day
  -- source: 'manual' today, 'rfid' once a reader is wired in later
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
