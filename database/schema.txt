CREATE TABLE tbl_schoolyear (
    schoolyear_id INT AUTO_INCREMENT PRIMARY KEY,
    year_start YEAR NOT NULL,
    year_end YEAR NOT NULL,
    schoolyear_name VARCHAR(50) NOT NULL,
    status TINYINT(1) NOT NULL DEFAULT 1,
    schoolyear_remarks TINYINT(1) DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE tbl_office (
    office_id INT AUTO_INCREMENT PRIMARY KEY,
    office_name VARCHAR(100) NOT NULL,
    office_description TEXT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    office_remarks TINYINT(1) DEFAULT 1
) ENGINE=InnoDB;

CREATE TABLE tbl_level (
    level_id INT AUTO_INCREMENT PRIMARY KEY,
    level_name VARCHAR(100) NOT NULL,
    level_type VARCHAR(50) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    level_remarks TINYINT(1) DEFAULT 1
) ENGINE=InnoDB;

CREATE TABLE tbl_position ( 
    position_id INT AUTO_INCREMENT PRIMARY KEY, 
    position_title VARCHAR(100) NOT NULL, 
    description TEXT, 
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, 
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, 
    position_remarks TINYINT(1) NOT NULL DEFAULT 1 
)  ENGINE=InnoDB;

CREATE TABLE tbl_teacher (
    teacher_id INT AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    middle_name VARCHAR(100),
    last_name VARCHAR(100) NOT NULL,
    extension_name VARCHAR(20),
    nick_name VARCHAR(100),
    gender ENUM('Male', 'Female', 'Other') NOT NULL,
    birthday DATE NOT NULL,
    phone_number VARCHAR(20),
    email VARCHAR(100) UNIQUE,
    position_id INT NOT NULL,
    department_id INT NOT NULL,
    date_hired date,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    teacher_remarks TINYINT(1) not null default 1
) ENGINE=InnoDB;

CREATE TABLE tbl_section (
    section_id INT AUTO_INCREMENT PRIMARY KEY,
    level_id INT NOT NULL,
    section_name VARCHAR(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    KEY idx_level_id (level_id)
) ENGINE=InnoDB;

CREATE TABLE tbl_subject (
    subject_id INT AUTO_INCREMENT PRIMARY KEY,
    level_id INT NOT NULL,
    subject_name VARCHAR(100) NOT NULL,
    subject_code VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    KEY idx_level_id (level_id)
) ENGINE=InnoDB;

CREATE TABLE tbl_student (
    student_id INT AUTO_INCREMENT PRIMARY KEY,
    lrn VARCHAR(20) UNIQUE,
    last_name VARCHAR(100) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    middle_name VARCHAR(100),
    extension_name VARCHAR(20),
    middle_initial VARCHAR(2),
    birthday DATE,
    sex ENUM('Male', 'Female', 'Other'),
    cp_no VARCHAR(20),
    spoken_language VARCHAR(100),
    fb_name VARCHAR(100),
    esc_id_no VARCHAR(50),
    former_school VARCHAR(150),
    father_name VARCHAR(150),
    mother_name VARCHAR(150),
    contact_person VARCHAR(150),
    contact_fb_name VARCHAR(100),
    street_name VARCHAR(150),
    barangay VARCHAR(100),
    municipality VARCHAR(100),
    province VARCHAR(100),
    contact_cp_no VARCHAR(20),
    student_status VARCHAR(30) DEFAULT 'active',
    student_photo VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE tbl_schoolyear_section (
    schoolyear_section_id INT AUTO_INCREMENT PRIMARY KEY,
    schoolyear_id INT NOT NULL,
    level_id INT NOT NULL,
    section_name VARCHAR(100) NOT NULL,
    section_remarks TINYINT(1) NOT NULL DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uq_sy_level_section (schoolyear_id, level_id, section_name),
    KEY idx_sy_section_schoolyear (schoolyear_id),
    KEY idx_sy_section_level (level_id),
    CONSTRAINT fk_sy_section_schoolyear
        FOREIGN KEY (schoolyear_id)
        REFERENCES tbl_schoolyear(schoolyear_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT fk_sy_section_level
        FOREIGN KEY (level_id)
        REFERENCES tbl_level(level_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
) ENGINE=InnoDB;

CREATE TABLE tbl_assignments (
    assignment_id INT AUTO_INCREMENT PRIMARY KEY,
    section_id INT NOT NULL,
    subject_id INT NOT NULL,
    teacher_id INT NULL,
    schedule_info VARCHAR(255) NULL,
    room_no VARCHAR(50) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uq_section_subject (section_id, subject_id),
    KEY idx_assignment_section (section_id),
    KEY idx_assignment_subject (subject_id),
    KEY idx_assignment_teacher (teacher_id),
    CONSTRAINT fk_assignment_section
        FOREIGN KEY (section_id)
        REFERENCES tbl_section(section_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT fk_assignment_subject
        FOREIGN KEY (subject_id)
        REFERENCES tbl_subject(subject_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    CONSTRAINT fk_assignment_teacher
        FOREIGN KEY (teacher_id)
        REFERENCES tbl_teacher(teacher_id)
        ON DELETE SET NULL
        ON UPDATE CASCADE
) ENGINE=InnoDB;
